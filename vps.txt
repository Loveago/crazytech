1. DNS CONFIGURATION (BOTH DOMAINS POINT TO SAME VPS IP)
================================================================================
A VPS has one public IP address. Both domains will point to that SAME IP.
When a visitor visits cmtshop.net, Nginx proxies to Next.js with Host: cmtshop.net,
serving customer storefronts. When a user visits crazytech.net, they access the portal.
When a visitor visits wolfdatahub.xyz, Nginx proxies to Next.js with Host: wolfdatahub.xyz,
serving customer storefronts. When a user visits crazytechent.com, they access the portal.

In your DNS manager (Namecheap, Cloudflare, GoDaddy, Hostinger, etc.):

For crazytech.net:
For crazytechent.com:
| Type | Host / Name | Value / Destination | TTL |
| :--- | :--- | :--- | :--- |
| A    | @           | <YOUR_VPS_IP>       | Auto / 1 min |
| A    | www         | <YOUR_VPS_IP>       | Auto / 1 min |

For cmtshop.net:
For wolfdatahub.xyz:
| Type | Host / Name | Value / Destination | TTL |
| :--- | :--- | :--- | :--- |
| A    | @           | <YOUR_VPS_IP>       | Auto / 1 min |
| A    | www         | <YOUR_VPS_IP>       | Auto / 1 min |

(Both domains live together on the same server and database!)

2. STEP 1: CONNECT TO VPS & CONFIGURE FIREWALL
================================================================================
# SSH into your VPS (replace your_vps_ip with your actual VPS IP address)
ssh root@your_vps_ip

# Update system packages and install prerequisites
sudo apt update && sudo apt upgrade -y
sudo apt install -y curl git ufw build-essential libpq-dev nginx certbot python3-certbot-nginx

# Configure UFW Firewall (allows SSH, HTTP port 80, HTTPS port 443)
sudo ufw allow OpenSSH
sudo ufw allow 'Nginx Full'
sudo ufw --force enable
sudo ufw status

4. STEP 3: INSTALL & CONFIGURE POSTGRESQL
================================================================================
# Install PostgreSQL and contrib tools
sudo apt install -y postgresql postgresql-contrib

# Start and enable PostgreSQL service
sudo systemctl start postgresql
sudo systemctl enable postgresql

# Create database user and database
# (Replace 'YourSecurePassword123!' with your chosen secure database password)
sudo -u postgres psql << 'EOF'
CREATE USER crazytech_user WITH PASSWORD 'YourSecurePassword123crazy!';
CREATE DATABASE crazytech_db OWNER crazytech_user;
GRANT ALL PRIVILEGES ON DATABASE crazytech_db TO crazytech_user;
ALTER USER crazytech_user CREATEDB;
\q
EOF

# Grant schema privileges (PostgreSQL 15+)
sudo -u postgres psql -d crazytech_db -c "GRANT ALL ON SCHEMA public TO crazytech_user;"

# Test connecting to the local database
PGPASSWORD='YourSecurePassword123crazy!' psql -U crazytech_user -d crazytech_db -h 127.0.0.1 -c "\conninfo"

8. STEP 7: CONFIGURE PM2 PROCESS MANAGER
================================================================================
# Create PM2 ecosystem file
cat << 'EOF' > /var/www/crazytech/ecosystem.config.js
module.exports = {
  apps: [
    {
      name: "crazytech",
      cwd: "/var/www/crazytech",
      script: "node_modules/next/dist/bin/next",
      args: "start -p 3000",
      instances: "max",
      exec_mode: "cluster",
      autorestart: true,
      watch: false,
      max_memory_restart: "1G",
      env: {
        NODE_ENV: "production",
        PORT: 3000
      }
    }
  ]
};
EOF

# Start application with PM2
pm2 start /var/www/crazytech/ecosystem.config.js

# Save PM2 process list to persist across reboots
pm2 save

# Verify application responds locally
curl -I http://127.0.0.1:3000

9. STEP 8: CONFIGURE NGINX FOR BOTH DOMAINS
================================================================================
We create two modular Nginx site configs: one for crazytechent.com and one for wolfdatahub.xyz.

--- [ 8A: NGINX CONFIG FOR CRAZYTECHENT.COM ] ---
cat << 'EOF' > /etc/nginx/sites-available/crazytechent.com
server {
    listen 80;
    listen [::]:80;
    server_name crazytechent.com www.crazytechent.com;

    client_max_body_size 250M;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;

        # WebSocket support
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Multi-tenant proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;

        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 120s;
        proxy_connect_timeout 60s;
    }

    # Static file caching
    location /_next/static/ {
        alias /var/www/crazytech/.next/static/;
        expires 365d;
        access_log off;
    }
}
EOF

--- [ 8B: NGINX CONFIG FOR WOLFDATAHUB.XYZ ] ---
cat << 'EOF' > /etc/nginx/sites-available/wolfdatahub.xyz
server {
    listen 80;
    listen [::]:80;
    server_name wolfdatahub.xyz www.wolfdatahub.xyz;

    client_max_body_size 250M;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;

        # WebSocket support
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Multi-tenant proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;

        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 120s;
        proxy_connect_timeout 60s;
    }

    # Static file caching
    location /_next/static/ {
        alias /var/www/crazytech/.next/static/;
        expires 365d;
        access_log off;
    }
}
EOF

--- [ 8C: ENABLE SITES & TEST NGINX ] ---
sudo ln -sf /etc/nginx/sites-available/crazytechent.com /etc/nginx/sites-enabled/
sudo ln -sf /etc/nginx/sites-available/wolfdatahub.xyz /etc/nginx/sites-enabled/
sudo rm -f /etc/nginx/sites-enabled/default

# Test Nginx syntax
sudo nginx -t

# Reload Nginx
sudo systemctl reload nginx
